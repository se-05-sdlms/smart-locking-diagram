%% =========================================================
%% BOUNDARY
%% =========================================================

class ParcelsController {
    <<boundary>>
    -IParcelService parcelService
    -IParcelPickupService parcelPickupService
    +GetParcels(view, search, from, to) ParcelListItemResponse[]
    +GetParcel(id) ParcelDetailResponse
    +GetHistory(id) ParcelStatusHistoryResponse[]
    +UnlockPickup(id) PickupUnlockResponse
    +ConfirmPickup(id, request) PickupConfirmationResponse
}

%% =========================================================
%% APPLICATION INTERFACES
%% =========================================================

class IParcelService {
    <<interface>>
    +GetParcelsAsync(userId, role, view, search, from, to, ct) ParcelListItemResponse[]
    +GetParcelAsync(userId, role, parcelId, ct) ParcelDetailResponse
    +GetHistoryAsync(userId, role, parcelId, ct) ParcelStatusHistoryResponse[]
}

class IParcelPickupService {
    <<interface>>
    +UnlockAsync(residentUserId, parcelId, ipAddress, deviceContext, ct) PickupUnlockResponse
    +ConfirmPickupAsync(residentUserId, lockerAccessEventId, ct) PickupConfirmationResponse
}

class ILockerAccessService {
    <<interface>>
    +OpenAsync(request, ct) OpenLockerResponse
}

class ILockerCommandDispatcher {
    <<interface>>
    +DispatchUnlockAsync(command, ct) void
}

%% =========================================================
%% SERVICE IMPLEMENTATIONS
%% =========================================================

class ParcelService {
    <<control>>
    -ApplicationDbContext dbContext
    +GetParcelsAsync(userId, role, view, search, from, to, ct) ParcelListItemResponse[]
    +GetParcelAsync(userId, role, parcelId, ct) ParcelDetailResponse
    +GetHistoryAsync(userId, role, parcelId, ct) ParcelStatusHistoryResponse[]
    -ScopeToUser(userId, role) IQueryable~Parcel~
}

class ParcelPickupService {
    <<control>>
    -ApplicationDbContext dbContext
    -ILockerAccessService lockerAccessService
    -TimeProvider timeProvider
    +UnlockAsync(residentUserId, parcelId, ipAddress, deviceContext, ct) PickupUnlockResponse
    +ConfirmPickupAsync(residentUserId, lockerAccessEventId, ct) PickupConfirmationResponse
}

class LockerAccessService {
    <<control>>
    -ApplicationDbContext dbContext
    -ILockerCommandDispatcher commandDispatcher
    -TimeProvider timeProvider
    +OpenAsync(request, ct) OpenLockerResponse
    -GetBlockedReason(compartment) string
    -ValidateContext(request) void
}

class MqttLockerCommandDispatcher {
    <<control>>
    -IConfiguration configuration
    +DispatchUnlockAsync(command, ct) void
}

class MqttLockerListenerService {
    <<control>>
    -IServiceScopeFactory serviceScopeFactory
    +ExecuteAsync(stoppingToken) void
    +HandleMessageAsync(topic, payload, ct) void
}

class ApplicationDbContext {
    <<persistence>>
    +Parcels
    +ParcelStatusHistories
    +LockerAccessEvents
    +Lockers
    +LockerCompartments
    +SystemPolicies
    +SaveChangesAsync(ct) int
}

%% =========================================================
%% RELATIONSHIPS - theo đúng luồng layer trái sang phải
%% =========================================================

ParcelsController --> IParcelService : delegates
ParcelsController --> IParcelPickupService : delegates

IParcelService <|.. ParcelService : implements
IParcelPickupService <|.. ParcelPickupService : implements
ILockerAccessService <|.. LockerAccessService : implements
ILockerCommandDispatcher <|.. MqttLockerCommandDispatcher : implements

ParcelPickupService --> ILockerAccessService : requests unlock
LockerAccessService --> ILockerCommandDispatcher : dispatches command

ParcelService --> ApplicationDbContext : reads
ParcelPickupService --> ApplicationDbContext : reads and writes
LockerAccessService --> ApplicationDbContext : reads and writes
MqttLockerListenerService --> ApplicationDbContext : updates door status

LockerAccessService ..> OpenLockerRequest : uses
LockerAccessService ..> OpenLockerResponse : returns
LockerAccessService ..> LockerUnlockCommand : creates

ParcelsController ..> ConfirmPickupRequest : receives
ParcelsController ..> PickupUnlockResponse : returns
ParcelsController ..> PickupConfirmationResponse : returns
ParcelsController ..> ParcelListItemResponse : returns
ParcelsController ..> ParcelDetailResponse : returns
ParcelsController ..> ParcelStatusHistoryResponse : returns

ApplicationDbContext ..> Parcel : manages
ApplicationDbContext ..> ParcelStatusHistory : manages
ApplicationDbContext ..> LockerAccessEvent : manages
ApplicationDbContext ..> Locker : manages
ApplicationDbContext ..> LockerCompartment : manages
ApplicationDbContext ..> SystemPolicy : manages

Parcel "1" *-- "0..*" ParcelStatusHistory : records
Parcel "1" *-- "0..*" LockerAccessEvent : records
Parcel "1" --> "0..1" OverdueCharge : has
Locker "1" *-- "1..*" LockerCompartment : contains
